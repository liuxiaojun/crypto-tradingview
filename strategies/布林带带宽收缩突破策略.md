//@version=5
strategy("布林带带宽收缩突破策略", overlay=true, initial_capital=5000, default_qty_type=strategy.percent_of_equity, default_qty_value=50, commission_type=strategy.commission.percent, commission_value=0.1,process_orders_on_close = true, calc_on_every_tick = false, calc_on_order_fills = false)
 
// ==================== 策略参数设置 ====================
// 布林带参数 - 针对ETH 30分钟优化
bb_length = input.int(24, "布林带周期", minval=1, tooltip="30分钟图建议24(12小时)或48(24小时)")
bb_mult = input.float(2.2, "标准差倍数", minval=0.1, step=0.1, tooltip="加密货币波动大,适当增加到2.2-2.5")
 
// 带宽收缩参数 - 优化增加交易频率
bandwidth_threshold = input.float(0.12, "带宽收缩阈值", minval=0.01, maxval=0.5, step=0.01, tooltip="提高到0.12-0.15增加交易机会")
bandwidth_lookback = input.int(6, "带宽比较周期", minval=5, maxval=100, tooltip="缩短到12小时(24根30分钟K线)")
squeeze_bars = input.int(2, "收缩持续K线数", minval=1, maxval=10, tooltip="减少到2-3根增加灵敏度")

// 突破确认参数
breakout_type = input.string("双向突破", "突破方向", options=["向上突破", "向下突破", "双向突破"], tooltip="ETH双向波动明显,建议双向")
volume_multiplier = input.float(0.5, "突破成交量倍数", minval=0, maxval=3.0, step=0.1, tooltip="降低到1.2-1.3增加信号")
use_middle_breakout = input.bool(true, "中轨突破确认", tooltip="价格需突破中轨才开仓")

// 止损止盈参数 - 简化为固定止损+跟踪止损
stop_loss_pct = input.float(2.5, "固定止损百分比%", minval=0.5, maxval=10, step=0.5, tooltip="紧急止损,防止大幅亏损")
trail_points_pct = input.float(3.0, "跟踪止损点数%", minval=0.5, maxval=10, step=0.5, tooltip="价格向有利方向移动多少激活跟踪")
trail_offset_pct = input.float(2.0, "跟踪止损偏移%", minval=0.5, maxval=10, step=0.5, tooltip="从最高点回撤多少触发止损")

// ==================== 布林带计算 ====================
[bb_middle, bb_upper, bb_lower] = ta.bb(close, bb_length, bb_mult)

// ==================== 带宽计算 ====================
// Bandwidth = (上轨 - 下轨) / 中轨
bandwidth = (bb_upper - bb_lower) / bb_middle

// 计算带宽的移动平均(用于识别收缩)
bandwidth_ma = ta.sma(bandwidth, bandwidth_lookback)

// 带宽是否处于历史低位
bandwidth_lowest = ta.lowest(bandwidth, bandwidth_lookback)
is_bandwidth_low = bandwidth <= bandwidth_lowest * 1.1 // 当前带宽接近最低值

// ==================== 收缩检测 ====================
// 带宽持续收缩
bandwidth_decreasing = true
for i = 1 to squeeze_bars
    if bandwidth[i-1] <= bandwidth[i]
        bandwidth_decreasing := false
        break

// 带宽处于收缩状态
is_squeezed = bandwidth < bandwidth_threshold and bandwidth < bandwidth_ma and is_bandwidth_low

// 收缩持续时间(连续多少根K线处于收缩状态)
var int squeeze_duration = 0
if is_squeezed
    squeeze_duration += 1
else
    squeeze_duration := 0

// ==================== 成交量确认 ====================
volume_ma = ta.sma(volume, 20)
volume_surge = volume > volume_ma * volume_multiplier

// ==================== 突破信号 ====================
// 向上突破:价格突破中轨且收缩后
breakout_up = false
if use_middle_breakout
    breakout_up := ta.crossover(close, bb_middle) and squeeze_duration >= squeeze_bars and volume_surge
else
    breakout_up := close > bb_middle and close[1] <= bb_middle[1] and squeeze_duration >= squeeze_bars and volume_surge

// 向下突破:价格跌破中轨且收缩后
breakout_down = false
if use_middle_breakout
    breakout_down := ta.crossunder(close, bb_middle) and squeeze_duration >= squeeze_bars and volume_surge
else
    breakout_down := close < bb_middle and close[1] >= bb_middle[1] and squeeze_duration >= squeeze_bars and volume_surge

// ==================== 出场条件 ====================
// 已移除所有手动出场逻辑,只保留跟踪止损+固定止损

// ==================== 交易逻辑 ====================
// 买入条件(做多)
long_condition = false
if breakout_type == "向上突破" or breakout_type == "双向突破"
    long_condition := breakout_up

// 做空条件
short_condition = false
if breakout_type == "向下突破" or breakout_type == "双向突破"
    short_condition := breakout_down

// 执行做多
if long_condition and strategy.position_size == 0
    strategy.entry("做多", strategy.long)
    
    // 计算止损价格和跟踪止损参数
    emergencyStopLong = close * (1 - stop_loss_pct / 100)
    trailPointsLong = close * trail_points_pct / 100
    trailOffsetLong = close * trail_offset_pct / 100
    
    // 设置跟踪止损+固定止损
    strategy.exit("做多出场", from_entry="做多", 
                  stop=emergencyStopLong,
                  trail_points=trailPointsLong, 
                  trail_offset=trailOffsetLong)

// 执行做空
if short_condition and strategy.position_size == 0
    strategy.entry("做空", strategy.short)
    
    // 计算止损价格和跟踪止损参数
    emergencyStopShort = close * (1 + stop_loss_pct / 100)
    trailPointsShort = close * trail_points_pct / 100
    trailOffsetShort = close * trail_offset_pct / 100
    
    // 设置跟踪止损+固定止损
    strategy.exit("做空出场", from_entry="做空", 
                  stop=emergencyStopShort,
                  trail_points=trailPointsShort, 
                  trail_offset=trailOffsetShort)

// ==================== 可视化 ====================
// 绘制布林带
upper_plot = plot(bb_upper, "上轨", color=color.red, linewidth=2)
middle_plot = plot(bb_middle, "中轨", color=color.blue, linewidth=2)
lower_plot = plot(bb_lower, "下轨", color=color.green, linewidth=2)

// 填充布林带区域
fill(upper_plot, lower_plot, color=color.new(color.blue, 95), title="布林带区域")

// 收缩状态背景着色
bgcolor(is_squeezed ? color.new(color.yellow, 90) : na, title="收缩状态")

// 绘制信号
plotshape(long_condition, title="做多信号", style=shape.triangleup, location=location.belowbar, color=color.new(color.green, 0), size=size.normal)
plotshape(short_condition, title="做空信号", style=shape.triangledown, location=location.abovebar, color=color.new(color.red, 0), size=size.normal)

// ==================== 带宽指标窗口 ====================
// 在独立窗口显示带宽
bandwidth_plot = plot(bandwidth, "带宽", color=color.purple, linewidth=2)
threshold_plot = plot(bandwidth_threshold, "收缩阈值", color=color.orange, linewidth=1, style=plot.style_line)

// 填充收缩区域
fill(bandwidth_plot, threshold_plot, color=bandwidth < bandwidth_threshold ? color.new(color.yellow, 85) : na, title="收缩区")

// 绘制带宽均线
plot(bandwidth_ma, "带宽均线", color=color.gray, linewidth=1, style=plot.style_circles)

// ==================== 信息面板 ====================
if barstate.islast
    var table info_table = table.new(position.top_right, 2, 9, border_width=1)
    
    table.cell(info_table, 0, 0, "指标", bgcolor=color.new(color.purple, 70), text_color=color.white)
    table.cell(info_table, 1, 0, "数值", bgcolor=color.new(color.purple, 70), text_color=color.white)
    
    table.cell(info_table, 0, 1, "带宽", text_color=color.white)
    table.cell(info_table, 1, 1, str.tostring(bandwidth, "#.####"), 
               bgcolor=is_squeezed ? color.new(color.yellow, 70) : color.new(color.gray, 80), 
               text_color=color.black)
    
    table.cell(info_table, 0, 2, "收缩状态", text_color=color.white)
    squeeze_status = is_squeezed ? "? 收缩中" : "正常"
    table.cell(info_table, 1, 2, squeeze_status, 
               bgcolor=is_squeezed ? color.new(color.yellow, 50) : color.new(color.gray, 80), 
               text_color=color.black)
    
    table.cell(info_table, 0, 3, "收缩时长", text_color=color.white)
    table.cell(info_table, 1, 3, str.tostring(squeeze_duration) + " K线", text_color=color.white)
    
    table.cell(info_table, 0, 4, "带宽均值", text_color=color.white)
    table.cell(info_table, 1, 4, str.tostring(bandwidth_ma, "#.####"), text_color=color.white)
    
    table.cell(info_table, 0, 5, "成交量状态", text_color=color.white)
    vol_status = volume_surge ? "放量" : "正常"
    table.cell(info_table, 1, 5, vol_status, 
               bgcolor=volume_surge ? color.new(color.green, 70) : color.new(color.gray, 80), 
               text_color=color.white)
    
    table.cell(info_table, 0, 6, "价格位置", text_color=color.white)
    price_pos = close > bb_middle ? "中轨上" : close < bb_middle ? "中轨下" : "中轨"
    table.cell(info_table, 1, 6, price_pos, text_color=color.white)
    
    table.cell(info_table, 0, 7, "固定止损", text_color=color.white)
    table.cell(info_table, 1, 7, str.tostring(stop_loss_pct, "#.#") + "%", text_color=color.white)
    
    table.cell(info_table, 0, 8, "跟踪止损", text_color=color.white)
    table.cell(info_table, 1, 8, str.tostring(trail_offset_pct, "#.#") + "%", text_color=color.white)
    
    table.cell(info_table, 0, 9, "当前信号", text_color=color.white)
    current_signal = long_condition ? "做多" : short_condition ? "做空" : "等待"
    signal_color = long_condition ? color.new(color.green, 50) : short_condition ? color.new(color.red, 50) : color.new(color.gray, 80)
    table.cell(info_table, 1, 9, current_signal, bgcolor=signal_color, text_color=color.white)

// ==================== 标注重要价位 ====================
// 标注收缩期间的价格范围
var line squeeze_line = na
if is_squeezed and not is_squeezed[1]
    squeeze_line := line.new(bar_index, bb_upper, bar_index, bb_lower, color=color.yellow, width=2, style=line.style_dotted)

// ==================== 警报设置 ====================
alertcondition(is_squeezed and not is_squeezed[1], title="带宽收缩警报", message="{{ticker}} 布林带开始收缩,准备突破!")
alertcondition(long_condition, title="向上突破警报", message="{{ticker}} 突破中轨向上,做多信号!")
alertcondition(short_condition, title="向下突破警报", message="{{ticker}} 突破中轨向下,做空信号!")
alertcondition(bandwidth > bandwidth_threshold * 2, title="带宽扩张警报", message="{{ticker}} 带宽急剧扩张,波动加大!")
