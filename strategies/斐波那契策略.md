//@version=4
strategy("julian's fib version不加仓版——final",
     overlay=true, initial_capital=1000,
     default_qty_type=strategy.percent_of_equity, default_qty_value=95,
     pyramiding=1, calc_on_every_tick=false, process_orders_on_close=true,calc_on_order_fills = false)

 
// 参数设置
EntryMsg = '{"id":"{{strategy.order.id}}","action":"{{strategy.order.action}}","marketPosition":"{{strategy.market_position}}","prevMarketPosition":"{{strategy.prev_market_position}}","instrument":"{{ticker}}","signalToken":"iplt0PQRRRd2aGnDdCGEc1AuACOb7wEZLzYkP5CeRMl/spTaMcInqu27wGBhuEeZQoeIMEVtpvIaGH5Xdephmg==","timestamp":"{{timenow}}","maxLag":"300","investmentType":"percentage_balance","amount":"100"}'
ExitMsg = '{"id":"{{strategy.order.id}}","action":"{{strategy.order.action}}","marketPosition":"{{strategy.market_position}}","prevMarketPosition":"{{strategy.prev_market_position}}","instrument":"{{ticker}}","signalToken":"iplt0PQRRRd2aGnDdCGEc1AuACOb7wEZLzYkP5CeRMl/spTaMcInqu27wGBhuEeZQoeIMEVtpvIaGH5Xdephmg==","timestamp":"{{timenow}}","maxLag":"300","investmentType":"percentage_position","amount":"100"}'

len      = input(14,  minval=1,   title="VWMA / 标准差 计算周期")
mult     = input(6.5,  minval=0.1, maxval=50, step=0.5, title="标准差倍数")
src      = input(hlc3, title="计算价格源")

slopeLB  = input(6,    minval=1,   title="lookbacknum")
basislen  = input(14,    minval=1,   title="basislen")
allowLongs  = input(true,  "允许做多")
allowShorts = input(true,  "允许做空")

// 加仓参数（最多一次加仓）
enableScaleIn = input(false,  "允许一次性加仓")
scaleBandLong = input(0.764, "多头加仓触发Fib比例", step=0.001, minval=0.236, maxval=1.0)
scaleBandShort= input(0.764, "空头加仓触发Fib比例", step=0.001, minval=0.236, maxval=1.0)
addParam =input(1.5,  minval=0, maxval=5, step=0.05, title="addParam")

// 止盈止损参数
useHardSL  = input(true,  "启用硬止损 (0.382 Fib)")
useTP      = input(true,  "启用止盈 (0.618 Fib)")

// 动态止盈参数（触及0.764后启动）
useTrail   = input(false,  "触及0.764后启用动态止盈")
trailMul   = input(0.5,   step=0.1, minval=0, maxval=3.0,title="trailMul")
trailParam =input(0.764,  minval=0, maxval=3, step=0.05, title="trailParam")


slParam =input(1,  minval=0, maxval=3, step=0.1, title="slParam")
lpParam =input(0.7,  minval=0, maxval=3, step=0.1, title="lpParam")
// 核心序列计算

// 中轨：VWMA (成交量加权移动均线)
basis = vwma(src, basislen)
// 标准差
dev   = mult * stdev(src, len)

// 按斐波那契比例划分上下带
upper_236 = basis + 0.236*dev
upper_382 = basis + 0.382*dev
upper_500 = basis + 0.5  *dev
upper_618 = basis + 0.618*dev
upper_764 = basis + 0.764*dev
upper_100 = basis + 1.0  *dev

lower_236 = basis - 0.236*dev
lower_382 = basis - 0.382*dev
lower_500 = basis - 0.5  *dev
lower_618 = basis - 0.618*dev
lower_764 = basis - 0.764*dev
lower_100 = basis - 1.0  *dev

// 基线斜率过滤
basisUp   = basis > nz(basis[slopeLB], basis)
basisDown = basis < nz(basis[slopeLB], basis)


// 确认突破
upper_break = close > upper_236
lower_break = close < lower_236


// 入场信号（收盘确认）

// longSignal  = allowLongs  and upper_break and basisUp
// shortSignal = allowShorts and lower_break and basisDown

longSignal  = allowLongs  and upper_break and basisUp
shortSignal = allowShorts and lower_break and basisDown 
// ====================
// 锁定入场当根参考值（防止后续漂移）
// ====================
var float refBasisLong = na
var float refDevLong   = na
var bool  scaledLong   = false

var float refBasisShort= na
var float refDevShort  = na
var bool  scaledShort  = false

// 追踪止损变量
var float longTrailStop = na
var bool  longHit764    = false
var float shortTrailStop = na
var bool  shortHit764    = false

// var bool firstOpenL = false
// var bool firstOpenS = false

// 入场时锁定参考
if (longSignal)
    // strategy.close("Short",alert_message = ExitMsg)
    strategy.entry("Long", strategy.long,alert_message = EntryMsg)
    refBasisLong := valuewhen(longSignal, basis, 0)
    refDevLong   := valuewhen(longSignal, dev,   0)
    scaledLong   := false
    longTrailStop := na
    longHit764    := false

    // firstOpenL := true

if (shortSignal )
    // strategy.close("Long",alert_message = ExitMsg)
    strategy.entry("Short", strategy.short,alert_message = EntryMsg)
    refBasisShort := valuewhen(shortSignal, basis, 0)
    refDevShort   := valuewhen(shortSignal, dev,   0)
    scaledShort   := false
    shortTrailStop := na
    shortHit764    := false

    // firstOpenS := true

// 空仓时重置
if strategy.position_size == 0
    refBasisLong := na
    refDevLong := na
    scaledLong := false
    longTrailStop := na
    longHit764 := false
    refBasisShort := na
    refDevShort := na
    scaledShort := false
    shortTrailStop := na
    shortHit764 := false

// ====================
// 一次性加仓逻辑
// ====================
canScaleLong  = enableScaleIn and strategy.position_size > 0 and not scaledLong and allowLongs
canScaleShort = enableScaleIn and strategy.position_size < 0 and not scaledShort and allowShorts



scaleTriggerLong  = canScaleLong  and close > (basis + addParam*dev)
scaleTriggerShort = canScaleShort and close < (basis - addParam*dev)



if (scaleTriggerLong)
    strategy.entry("LongADDDDDDDDD", strategy.long,comment = "LongADDDDDDDDD",alert_message = EntryMsg)
    scaledLong := true

    // firstOpenL :=false

if (scaleTriggerShort)
    strategy.entry("ShortADDDDDDDDD", strategy.short,comment = "ShortADDDDDDDDD",alert_message = EntryMsg)
    scaledShort := true

    // firstOpenS :=false

// ====================
// 退出逻辑：使用 strategy.close 替代 strategy.exit
// ====================

// 多头退出
if strategy.position_size > 0 and not na(refBasisLong) and not na(refDevLong)
    longSL = refBasisLong - slParam*refDevLong
    longTP = refBasisLong + lpParam*refDevLong
   
    
    // 检查是否触及 0.764
    if high >= (refBasisLong + trailParam*refDevLong)
        longHit764 := true
    
    // 更新追踪止损
    if useTrail and longHit764
        trailOffset = trailMul * refDevLong
        longTrailStop := na(longTrailStop) ? high - trailOffset : max(longTrailStop, high - trailOffset)
    
    // 硬止损
    if useHardSL and close <= longSL
        strategy.close_all(comment = "Long SL",alert_message = ExitMsg)
    
    // 止盈
    if useTP and close >= longTP
        strategy.close_all(comment = "Long TP",alert_message = ExitMsg)
    
    // 追踪止损
    if useTrail and longHit764 and not na(longTrailStop) and close <= longTrailStop
        strategy.close_all(comment = "Long Trail",alert_message = ExitMsg)

// 空头退出
if strategy.position_size < 0 and not na(refBasisShort) and not na(refDevShort)
    shortSL = refBasisShort + slParam*refDevShort
    shortTP = refBasisShort - lpParam*refDevShort

    
    // 检查是否触及 0.764
    if low <= (refBasisShort - trailParam*refDevShort)
        shortHit764 := true
    
    // 更新追踪止损
    if useTrail and shortHit764
        trailOffset = trailMul * refDevShort
        shortTrailStop := na(shortTrailStop) ? low + trailOffset : min(shortTrailStop, low + trailOffset)
    
    // 硬止损
    if useHardSL and close >= shortSL
        strategy.close_all(comment = "Short SL",alert_message = ExitMsg)
    
    // 止盈
    if useTP and close <= shortTP
        strategy.close_all(comment = "Short TP",alert_message = ExitMsg)
    
    // 追踪止损
    if useTrail and shortHit764 and not na(shortTrailStop) and close >= shortTrailStop
        strategy.close_all(comment = "Short Trail",alert_message = ExitMsg)

// ====================
// 绘图与信号标记
// ====================
// 中轨
plot(basis, color=color.fuchsia, linewidth=2, title="VWMA 基线")
// 上方斐波带
plot(upper_236, color=color.gray, linewidth=1, title="U 0.236")
plot(upper_382, color=color.gray, linewidth=1, title="U 0.382")
plot(upper_500, color=color.gray, linewidth=1, title="U 0.5")
plot(upper_618, color=color.gray, linewidth=1, title="U 0.618")
plot(upper_764, color=color.gray, linewidth=1, title="U 0.764")
plot(upper_100, color=color.red,   linewidth=2, title="U 1.0")
// 下方斐波带
plot(lower_236, color=color.gray, linewidth=1, title="L 0.236")
plot(lower_382, color=color.gray, linewidth=1, title="L 0.382")
plot(lower_500, color=color.gray, linewidth=1, title="L 0.5")
plot(lower_618, color=color.gray, linewidth=1, title="L 0.618")
plot(lower_764, color=color.gray, linewidth=1, title="L 0.764")
plot(lower_100, color=color.green, linewidth=2, title="L 1.0")

// 信号标记
plotshape(longSignal,  title="多头信号", text="LONG",  style=shape.labelup,   location=location.belowbar, color=color.green, textcolor=color.white, size=size.tiny)
plotshape(shortSignal, title="空头信号", text="SHORT", style=shape.labeldown, location=location.abovebar, color=color.red,   textcolor=color.white, size=size.tiny)

// 告警
alertcondition(longSignal,  title="多头入场",  message="FBB Long Entry")
alertcondition(shortSignal, title="空头入场", message="FBB Short Entry")

plot(refDevLong, color=color.blue, linewidth=2, title="refDevLong", trackprice=true)
