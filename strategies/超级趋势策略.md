//@version=4
strategy("julian's trends 多空完整版",
     overlay=true, initial_capital=1000,
     default_qty_type=strategy.percent_of_equity, default_qty_value=95,
     pyramiding=0, calc_on_every_tick=false, process_orders_on_close=true,calc_on_order_fills = false)

//====================
// 参数
//====================
Periods      = input(title="ATR Period", type=input.integer, defval=46, minval=1)
Multiplier   = input(title="Supertrend Multiplier", type=input.float, defval=1.4, step=0.1)
useWilderATR = input(title="Use Wilder ATR (rma) ?  (off=SMA(TR))", type=input.bool, defval=true)

allowLongs   = input(true,  "Allow Longs")
allowShorts  = input(true,  "Allow Shorts")

// 风控（以 ATR 为单位）
useSL        = input(true,  "Enable ATR Stop Loss")
slATR        = input(title="SL ATR Multiple", type=input.float, defval=1.7, step=0.1, minval=0.1)

useTP        = input(false,  "Enable ATR Take Profit")
tpATR        = input(title="TP ATR Multiple", type=input.float, defval=3, step=0.1, minval=0.1)

// 可选：跟踪止损（以 ATR 为单位）
useTrail     = input(false, "Enable ATR Trailing Stop")
trailATR     = input(title="Trailing ATR Multiple", type=input.float, defval=1.5, step=0.1, minval=0.5)

// 视觉与告警
showSignals  = input(true,  "Show Buy/Sell Labels")
highlighting = input(true,  "Background Highlight")

//====================
// 兼容、基础序列
//====================
src = input(hl2, title="Price Source (Supertrend uses src)")
prevClose = nz(close[1], close)


// 明确 True Range（v4 中直接写出来，避免"tr"语义不清）
trueRange = max(high - low, max(abs(high - prevClose), abs(low - prevClose)))

// ATR 两种口径：Wilder(RMA) / SMA
atr_rma = rma(trueRange, Periods)
atr_sma = sma(trueRange, Periods)
atrUsed = useWilderATR ? atr_rma : atr_sma


basicUp   = src - Multiplier * atrUsed
basicDown = src + Multiplier * atrUsed

var float finalUp   = na
var float finalDown = na

up1 = nz(finalUp[1], basicUp)
dn1 = nz(finalDown[1], basicDown)
EntryMsg = '{"id": "{{strategy.order.id}}","action": "{{strategy.order.action}}","marketPosition": "{{strategy.market_position}}","prevMarketPosition": "{{strategy.prev_market_position}}","marketPositionSize": "{{strategy.market_position_size}}","prevMarketPositionSize": "{{strategy.prev_market_position_size}}","instrument": "{{ticker}}","signalToken": "zDPsSsyezKI4e8t8nKUMTDnAiYDK7fAV74Ng5VqnMyJEekZFVVJyvkEmOSPeBLkNpwORvhJ9Kwa3uZV9oMahSQ==","timestamp": "{{timenow}}","maxLag": "300","investmentType": "percentage_balance","amount": "100"}'
ExitMsg = '{"id": "{{strategy.order.id}}","action": "{{strategy.order.action}}","marketPosition": "{{strategy.market_position}}","prevMarketPosition": "{{strategy.prev_market_position}}","marketPositionSize": "{{strategy.market_position_size}}","prevMarketPositionSize": "{{strategy.prev_market_position_size}}","instrument": "{{ticker}}","signalToken": "zDPsSsyezKI4e8t8nKUMTDnAiYDK7fAV74Ng5VqnMyJEekZFVVJyvkEmOSPeBLkNpwORvhJ9Kwa3uZV9oMahSQ==","timestamp": "{{timenow}}","maxLag": "300","investmentType": "percentage_position","amount": "100"}'

// 在上行阶段，上轨只抬高不下移；在下行阶段，下轨只下移不抬高
finalUp   := prevClose > up1 ? max(basicUp, up1) : basicUp
finalDown := prevClose < dn1 ? min(basicDown, dn1) : basicDown

// 趋势方向：1 多头，-1 空头
var int trend = 1
trend := nz(trend[1], 1)
trend := trend == -1 and close > dn1 ?  1 :
         trend ==  1 and close < up1 ? -1 : trend

// 翻转信号（以收盘确认，不重绘）
buySignal  = trend ==  1 and trend[1] == -1
sellSignal = trend == -1 and trend[1] ==  1



//====================
// 策略下单：翻多,平空并做多；翻空,平多并做空
//====================
if (buySignal and allowLongs)
    // if(strategy.position_size<0)
    //     strategy.close("Short")

    strategy.entry("Long", strategy.long)



if (sellSignal and allowShorts)
    // if(strategy.position_size>0)
    //     strategy.close("Long")

    strategy.entry("Short", strategy.short)



//====================
// 使用 strategy.close 实现止损止盈和追踪止损
//====================
// 追踪止损需要记录最高/最低价
var float longTrailStop = na
var float shortTrailStop = na

if strategy.position_size > 0
    // 多单
    longEntry = strategy.position_avg_price
    
    // 更新追踪止损
    if useTrail
        trailOffset = trailATR * atrUsed
        longTrailStop := na(longTrailStop) ? close - trailOffset : max(longTrailStop, close - trailOffset)
    
    // 止损条件
    longSL = useSL ? longEntry - slATR * atrUsed : na
    longTP = useTP ? longEntry + tpATR * atrUsed : na
    
    // 止损
    if useSL and not na(longSL) and close <= longSL
        strategy.close("Long", comment = "Long SL")
    
    // 止盈
    if useTP and not na(longTP) and close >= longTP
        strategy.close("Long", comment = "Long TP")
    
    // 追踪止损
    if useTrail and not na(longTrailStop) and close <= longTrailStop
        strategy.close("Long", comment = "Long Trail")

if strategy.position_size <= 0
    longTrailStop := na

if strategy.position_size < 0
    // 空单
    shortEntry = strategy.position_avg_price
    
    // 更新追踪止损
    if useTrail
        trailOffset = trailATR * atrUsed
        shortTrailStop := na(shortTrailStop) ? close + trailOffset : min(shortTrailStop, close + trailOffset)
    
    // 止损条件
    shortSL = useSL ? shortEntry + slATR * atrUsed : na
    shortTP = useTP ? shortEntry - tpATR * atrUsed : na
    
    // 止损
    if useSL and not na(shortSL) and close >= shortSL
        strategy.close("Short", comment = "Short SL")
    
    // 止盈
    if useTP and not na(shortTP) and close <= shortTP
        strategy.close("Short", comment = "Short TP")
    
    // 追踪止损
    if useTrail and not na(shortTrailStop) and close >= shortTrailStop
        strategy.close("Short", comment = "Short Trail")

if strategy.position_size >= 0
    shortTrailStop := na

//====================
// 可视化
//====================
upPlot = plot(trend == 1 ? finalUp   : na, title="Up Trend",   style=plot.style_linebr, linewidth=2, color=color.green)
dnPlot = plot(trend == 1 ? na        : finalDown, title="Down Trend", style=plot.style_linebr, linewidth=2, color=color.red)

// 标签
plotshape(showSignals and buySignal  ? finalUp   : na, title="Buy",  text="Buy",  location=location.absolute, style=shape.labelup,   size=size.tiny, color=color.green, textcolor=color.white)
plotshape(showSignals and sellSignal ? finalDown : na, title="Sell", text="Sell", location=location.absolute, style=shape.labeldown, size=size.tiny, color=color.red,   textcolor=color.white)

// 高亮背景
mPlot = plot(ohlc4, title="", style=plot.style_circles, linewidth=0)
longFillColor  = highlighting ? (trend == 1 ? color.green : color.white) : color.white
shortFillColor = highlighting ? (trend == -1 ? color.red   : color.white) : color.white
fill(mPlot, upPlot, title="UpTrend Highlighter",   color=longFillColor)
fill(mPlot, dnPlot, title="DownTrend Highlighter", color=shortFillColor)

// 告警（可选）
alertcondition(buySignal,  title="SuperTrend Buy",  message="SuperTrend turned UP (Buy)")
alertcondition(sellSignal, title="SuperTrend Sell", message="SuperTrend turned DOWN (Sell)")
alertcondition(trend != trend[1], title="SuperTrend Direction Change", message="SuperTrend changed direction")
