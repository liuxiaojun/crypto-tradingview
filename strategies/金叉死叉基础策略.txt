//@version=5
strategy("基础金叉死叉策略", overlay=true, initial_capital=10000, default_qty_type=strategy.percent_of_equity, default_qty_value=10)

// ==================== 参数设置 ====================
fastLen = input.int(5,  "快线周期", minval=1)   // 短期均线，常用 5 或 10
slowLen = input.int(20, "慢线周期", minval=1)   // 长期均线，常用 20 或 60

// ==================== 计算均线 ====================
fastMA = ta.sma(close, fastLen)   // 快线：收盘价的简单移动平均
slowMA = ta.sma(close, slowLen)   // 慢线：收盘价的简单移动平均

// ==================== 金叉 / 死叉信号 ====================
// 金叉：快线从下方穿越慢线 → 做多
goldenCross = ta.crossover(fastMA, slowMA)

// 死叉：快线从上方穿越慢线 → 平多（本例不做空，只做多）
deathCross = ta.crossunder(fastMA, slowMA)

// ==================== 下单逻辑 ====================
if goldenCross
    strategy.entry("做多", strategy.long)   // 金叉开多

if deathCross
    strategy.close("做多")                  // 死叉平多

// ==================== 图表显示 ====================
plot(fastMA, "快线", color=color.new(color.blue, 0), linewidth=2)
plot(slowMA, "慢线", color=color.new(color.orange, 0), linewidth=2)

// 在 K 线上标记金叉、死叉
plotshape(goldenCross, title="金叉", style=shape.triangleup,   location=location.belowbar, color=color.green, size=size.small)
plotshape(deathCross,  title="死叉", style=shape.triangledown, location=location.abovebar, color=color.red,   size=size.small)
